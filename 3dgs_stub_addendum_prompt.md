## MISSION OF THIS ADDENDUM

The main app prompt already defines the full pipeline architecture.
This addendum instructs you to implement a **complete STUB / simulation system**
for RealityCapture and LichtFeld Studio, allowing the full wizard UI to be
developed, tested, and demoed **without any external tool installed or compiled**.

Every stub must be **indistinguishable from the real pipeline at the UI level**:
same WebSocket messages, same file outputs, same step transitions.
The only difference is the source of execution.

## 1 — STUB CONFIGURATION IN `config.py`

Add the following stub flags to `backend/core/config.py`.

and **persistable** in `config.json` at project root.

```python
# backend/core/config.py  (additions)

from pathlib import Path
from pydantic import BaseModel
from typing import Optional
import json

CONFIG_PATH = Path(__file__).parents[2] / "config.json"

class ToolPaths(BaseModel):
    rc_exe_path: Optional[str] = None
    lfs_exe_path: Optional[str] = None
    ffmpeg_path: str = "ffmpeg"
    blender_exe_path: Optional[str] = None
    supersplat_url: str = "https://superspl.at/editor"

class StubConfig(BaseModel):
    """
    Stub flags — set True to simulate a tool without it being installed.
    Each flag is independent: you can stub RC but run real FFmpeg, etc.
    """
    ffmpeg_stub: bool = False        # Simulate FFmpeg frame extraction
    rc_stub: bool = True             # Simulate RealityCapture alignment  ← DEFAULT ON
    lfs_stub: bool = True            # Simulate LichtFeld Studio training  ← DEFAULT ON
    blender_stub: bool = True        # Simulate Blender scene export        ← DEFAULT ON

    # Stub behavior tuning
    rc_stub_duration_seconds: float = 8.0    # Total simulated RC duration
    lfs_stub_duration_seconds: float = 15.0  # Total simulated LFS training duration
    lfs_stub_iterations: int = 30000         # Iterations to simulate
    lfs_stub_fake_ply: bool = True           # Copy sample.ply to output at end

class AppConfig(BaseModel):
    tools: ToolPaths = ToolPaths()
    stubs: StubConfig = StubConfig()

def load_config() -> AppConfig:
    if CONFIG_PATH.exists():
        return AppConfig(**json.loads(CONFIG_PATH.read_text()))
    return AppConfig()

def save_config(cfg: AppConfig):
    CONFIG_PATH.write_text(cfg.model_dump_json(indent=2))

# Singleton
app_config: AppConfig = load_config()
```

## 2 — STUB ASSET: `tools/test_assets/sample.ply`

Create `tools/test_assets/` directory and generate a **minimal valid binary PLY file**
that SuperSplat can open. The file must contain at least 100 3DGS Gaussians
with all required properties.

Write a generator script `tools/test_assets/generate_sample_ply.py`:

```python
"""
Generates a minimal valid 3D Gaussian Splatting .ply file for stub testing.
Run once: python tools/test_assets/generate_sample_ply.py
"""
import struct
import random
import math
from pathlib import Path

OUTPUT = Path(__file__).parent / "sample.ply"
N = 500  # number of Gaussians

def write_ply_header(f, n):
    header = f"""ply
format binary_little_endian 1.0
comment 3DGS stub sample — generated by 3dgs-pipeline-app
element vertex {n}
property float x
property float y
property float z
property float nx
property float ny
property float nz
property float f_dc_0
property float f_dc_1
property float f_dc_2
property float f_rest_0
property float f_rest_1
property float f_rest_2
property float opacity
property float scale_0
property float scale_1
property float scale_2
property float rot_0
property float rot_1
property float rot_2
property float rot_3
end_header\n"""
    f.write(header.encode("ascii"))

def write_ply_data(f, n):
    for i in range(n):
        # Random position in a 2m sphere
        theta = random.uniform(0, 2 * math.pi)
        phi = random.acos(random.uniform(-1, 1))
        r = random.uniform(0, 1.0)
        x = r * math.sin(phi) * math.cos(theta)
        y = r * math.sin(phi) * math.sin(theta)
        z = r * math.cos(phi)
        # Normal (unit, random)
        nx, ny, nz = 0.0, 0.0, 1.0
        # SH DC (color ~warm grey)
        f_dc_0 = random.uniform(0.3, 0.8)
        f_dc_1 = random.uniform(0.3, 0.7)
        f_dc_2 = random.uniform(0.2, 0.6)
        # SH rest (zeroed)
        f_rest = [0.0, 0.0, 0.0]
        # Opacity (logit ~0.8)
        opacity = 1.386  # sigmoid(1.386) ≈ 0.8
        # Scale (log scale, small ellipsoids)
        scale = [random.uniform(-4.0, -2.5)] * 3
        # Rotation quaternion (identity-ish)
        rot = [1.0, 0.0, 0.0, 0.0]

        values = [x, y, z, nx, ny, nz, f_dc_0, f_dc_1, f_dc_2,
                  *f_rest, opacity, *scale, *rot]
        f.write(struct.pack(f"{len(values)}f", *values))

with open(OUTPUT, "wb") as f:
    write_ply_header(f, N)
    write_ply_data(f, N)

print(f"[stub] Generated {N} Gaussians → {OUTPUT}")
print(f"[stub] File size: {OUTPUT.stat().st_size / 1024:.1f} KB")
```

**Run this script automatically** in `setup.py` after `.venv` creation:


```python
# ── Real RC runner (calls RealityScan.exe CLI) ─────────────────────────────
async def run_rc_real(project_path: Path, broadcast_fn, settings: dict):
    """Calls RealityScan.exe via CLI with the rscmd script."""
    rc_exe = Path(app_config.tools.rc_exe_path)
    if not rc_exe.exists():
        raise FileNotFoundError(
            f"RealityScan.exe not found at: {rc_exe}\n"
            "Enable rc_stub in Settings, or install RealityCapture via Epic Games Launcher."
        )
    frames_dir = project_path / "frames"
    rc_output = project_path / "rc_output"
    rc_output.mkdir(exist_ok=True)
    scripts_dir = Path("backend/scripts")

    cmd = [
        str(rc_exe),
        "-execrscmd", str(scripts_dir / "rc_align_export.rscmd"),
        str(frames_dir),
        str(rc_output),
        str(scripts_dir),
    ]
    await broadcast_fn("rc", "INFO", f"[RC] Launching: {' '.join(cmd)}")

    proc = await asyncio.create_subprocess_exec(
        *cmd,
        stdout=asyncio.subprocess.PIPE,
        stderr=asyncio.subprocess.STDOUT,
    )
    async for raw_line in proc.stdout:
        line = raw_line.decode("utf-8", errors="replace").strip()
        await broadcast_fn("rc", _classify_rc_line(line), line)

    await proc.wait()
    if proc.returncode != 0:
        raise RuntimeError(f"RealityCapture exited with code {proc.returncode}")

    await broadcast_fn("rc", "SUCCESS",
        "[RC] Alignment complete.", progress=1.0)


def _classify_rc_line(line: str) -> str:
    line_lower = line.lower()
    if "error" in line_lower or "failed" in line_lower:
        return "ERROR"
    if "warning" in line_lower:
        return "WARNING"
    if "aligned" in line_lower or "export" in line_lower or "done" in line_lower:
        return "SUCCESS"
    return "INFO"
```

## 5 — STUB: STEP 4 (LichtFeld Studio) — `step_lfs.py`

The most detailed stub — must simulate the live metrics that the frontend charts.

```python
# backend/core/steps/step_lfs.py

import asyncio
import shutil
import math
import random
from pathlib import Path
from backend.core.config import app_config


# ── Real LFS runner ─────────────────────────────────────────────────────────
async def run_lfs_real(project_path: Path, broadcast_fn, settings: dict):
    """Calls LichtFeld-Studio.exe in headless/CLI mode."""
    lfs_exe = Path(app_config.tools.lfs_exe_path)
    if not lfs_exe.exists():
        raise FileNotFoundError(
            f"LichtFeld-Studio executable not found at: {lfs_exe}\n"
            "Build LichtFeld Studio from source (C++23 + CUDA 12.8 required),\n"
            "then set lfs_exe_path in Settings, or enable lfs_stub."
        )
    rc_output = project_path / "rc_output"
    lfs_output = project_path / "lfs_output"
    lfs_output.mkdir(exist_ok=True)

    iterations = settings.get("iterations", 30000)
    strategy = settings.get("strategy", "default")
    strategy_flag = ["--strategy", "mcmc"] if strategy == "mcmc" else []

    cmd = [
        str(lfs_exe),
        "-d", str(rc_output),
        "-o", str(lfs_output),
        "-i", str(iterations),
        *strategy_flag,
        "--eval",
    ]
    await broadcast_fn("lfs", "INFO", f"[LFS] Launching: {' '.join(cmd)}")

    proc = await asyncio.create_subprocess_exec(
        *cmd,
        stdout=asyncio.subprocess.PIPE,
        stderr=asyncio.subprocess.STDOUT,
    )
    async for raw_line in proc.stdout:
        line = raw_line.decode("utf-8", errors="replace").strip()
        metrics = _parse_lfs_metrics(line)
        await broadcast_fn("lfs", "INFO", line,
            progress=metrics.get("progress"),
            data=metrics if metrics else None)

    await proc.wait()
    if proc.returncode != 0:
        raise RuntimeError(f"LichtFeld Studio exited with code {proc.returncode}")

    await broadcast_fn("lfs", "SUCCESS",
        "[LFS] Training complete.", progress=1.0)


def _parse_lfs_metrics(line: str) -> dict:
    """
    Parse LichtFeld Studio stdout lines for metrics.
    Expected format (approximate): 'iter 5000/30000 loss=0.0234 psnr=24.56 gaussians=182340'
    Adjust regex to match actual LFS output format.
    """
    import re
    metrics = {}
    m_iter = re.search(r"iter[ation]*\s+(\d+)\s*/\s*(\d+)", line, re.IGNORECASE)
    if m_iter:
        current, total = int(m_iter.group(1)), int(m_iter.group(2))
        metrics["iteration"] = current
        metrics["progress"] = current / total if total else 0
    m_loss = re.search(r"loss[=:\s]+([0-9.eE+\-]+)", line, re.IGNORECASE)
    if m_loss:
        metrics["loss"] = float(m_loss.group(1))
    m_psnr = re.search(r"psnr[=:\s]+([0-9.]+)", line, re.IGNORECASE)
    if m_psnr:
        metrics["psnr"] = float(m_psnr.group(1))
    m_gauss = re.search(r"gaussian[s]?[=:\s]+([0-9]+)", line, re.IGNORECASE)
    if m_gauss:
        metrics["num_gaussians"] = int(m_gauss.group(1))
    return metrics


# ── Stub LFS runner ─────────────────────────────────────────────────────────
async def run_lfs_stub(project_path: Path, broadcast_fn, settings: dict):
    """
    Simulates LichtFeld Studio 3DGS training with realistic metrics.
    Streams progressive log lines + metric data over WebSocket.
    Copies sample.ply to lfs_output/ at completion.
    """
    lfs_output = project_path / "lfs_output"
    lfs_output.mkdir(exist_ok=True)

    duration = app_config.stubs.lfs_stub_duration_seconds
    max_iter = settings.get("iterations", app_config.stubs.lfs_stub_iterations)
    strategy = settings.get("strategy", "default")

    await broadcast_fn("lfs", "INFO",
        "⚠️  [STUB MODE] LichtFeld Studio is simulated — no real exe called.",
        progress=0.0)
    await asyncio.sleep(0.3)

    # --- Startup phase
    startup_lines = [
        f"LichtFeld Studio (GPLv3) — C++23/CUDA build",
        f"Loading COLMAP dataset from: {project_path / 'rc_output'}",
        f"Found cameras.bin: OK | images.bin: OK | points3D.bin: OK",
        f"Images loaded: {settings.get('frame_count', 120)}",
        f"Sparse points: {random.randint(180000, 320000):,}",
        f"Strategy: {strategy.upper()} | Iterations: {max_iter:,}",
        f"CUDA device: NVIDIA GeForce RTX 3080 (10240 MB)",
        f"Initializing Gaussians from sparse point cloud...",
        f"Initial Gaussian count: {random.randint(90000, 130000):,}",
        "Starting training...",
    ]
    for line in startup_lines:
        await broadcast_fn("lfs", "INFO", line, progress=0.0)
        await asyncio.sleep(0.3)

    # --- Training simulation
    # Metrics model: realistic convergence curves
    # loss: starts ~0.35, exponential decay toward ~0.018
    # psnr: starts ~18dB, logarithmic rise toward ~28dB
    # gaussians: grows from ~100K to ~350K then stabilizes
    # iter/s: ~500–800 iter/s on RTX 3080

    checkpoint_intervals = [1000, 2000, 5000, 7000, 10000, 15000, 20000, 25000, max_iter]
    checkpoints = [c for c in checkpoint_intervals if c <= max_iter]
    if max_iter not in checkpoints:
        checkpoints.append(max_iter)

    total_sleep = duration - 3.0  # Reserve 3s for startup + shutdown
    sleep_per_checkpoint = total_sleep / len(checkpoints)

    prev_iter = 0
    for checkpoint_iter in checkpoints:
        await asyncio.sleep(sleep_per_checkpoint)

        # Compute simulated metrics at this checkpoint
        t = checkpoint_iter / max_iter  # 0.0 → 1.0

        # Loss: L(t) = 0.35 * exp(-3.5 * t) + 0.018 + noise
        loss = 0.35 * math.exp(-3.5 * t) + 0.018 + random.gauss(0, 0.002)
        loss = max(0.012, loss)

        # PSNR: P(t) = 18 + 10 * log10(1 + 8 * t) + noise
        psnr = 18.0 + 10.0 * math.log10(1 + 8 * t) + random.gauss(0, 0.15)
        psnr = min(32.0, max(16.0, psnr))

        # Gaussians: grows fast then plateau
        if t < 0.4:
            gaussians = int(100_000 + 600_000 * t)
        else:
            gaussians = int(340_000 + 20_000 * math.sin(t * math.pi))
        gaussians += random.randint(-5000, 5000)

        # Iter/s
        iters_in_block = checkpoint_iter - prev_iter
        iter_per_sec = iters_in_block / sleep_per_checkpoint if sleep_per_checkpoint > 0 else 500

        # Log line (mimics real LFS format)
        log_line = (
            f"iter {checkpoint_iter:>6}/{max_iter} | "
            f"loss={loss:.4f} | "
            f"psnr={psnr:.2f}dB | "
            f"gaussians={gaussians:,} | "
            f"speed={iter_per_sec:.0f}it/s"
        )
        metrics_data = {
            "iteration": checkpoint_iter,
            "loss": round(loss, 5),
            "psnr": round(psnr, 3),
            "num_gaussians": gaussians,
            "fps": round(iter_per_sec, 1),
        }

        await broadcast_fn("lfs", "INFO", f"[STUB] {log_line}",
            progress=t,
            data=metrics_data)

        # Densification event (every ~5k iterations early on)
        if checkpoint_iter <= 15000 and checkpoint_iter % 5000 == 0:
            await broadcast_fn("lfs", "INFO",
                f"[STUB] Densification step — added ~{random.randint(8000, 15000):,} Gaussians",
                data=metrics_data)

        # Checkpoint save
        if checkpoint_iter in [10000, 20000, max_iter]:
            ckpt_path = lfs_output / f"checkpoint_{checkpoint_iter}.pth"
            ckpt_path.touch()  # Stub: empty file as placeholder
            await broadcast_fn("lfs", "INFO",
                f"[STUB] Checkpoint saved → {ckpt_path.name}")

        prev_iter = checkpoint_iter

    # --- Export PLY
    await broadcast_fn("lfs", "INFO",
        "[STUB] Saving final scene as point_cloud.ply...", progress=0.98)
    await asyncio.sleep(0.5)

    stub_ply = Path("tools/test_assets/sample.ply")
    output_ply = lfs_output / "point_cloud.ply"
    if stub_ply.exists():
        shutil.copy(stub_ply, output_ply)
    else:
        # Fallback: create empty placeholder
        output_ply.write_bytes(b"ply\nformat binary_little_endian 1.0\nelement vertex 0\nend_header\n")

    # Also write a .splat stub (empty, for format detection testing)
    (lfs_output / "output.splat").write_bytes(b"")

    final_loss = 0.35 * math.exp(-3.5) + 0.018
    final_psnr = 18.0 + 10.0 * math.log10(9)
    final_gauss = 342_000 + random.randint(-10000, 10000)

    await broadcast_fn("lfs", "SUCCESS",
        f"[STUB] LichtFeld Studio training complete. "
        f"Final: loss={final_loss:.4f} | psnr={final_psnr:.2f}dB | "
        f"gaussians={final_gauss:,} | "
        f"PLY → {output_ply}",
        progress=1.0,
        data={
            "iteration": max_iter,
            "loss": round(final_loss, 5),
            "psnr": round(final_psnr, 3),
            "num_gaussians": final_gauss,
        })


# ── Dispatcher ──────────────────────────────────────────────────────────────
async def run_lfs(project_path: Path, broadcast_fn, settings: dict):
    if app_config.stubs.lfs_stub:
        await run_lfs_stub(project_path, broadcast_fn, settings)
    else:
        await run_lfs_real(project_path, broadcast_fn, settings)
```

```python
async def update_settings(data: dict):
    """Partial update — merges into current config."""
    current = app_config.model_dump()
    # Deep merge top-level keys
    for key, val in data.items():
        if key in current and isinstance(current[key], dict) and isinstance(val, dict):
            current[key].update(val)
        else:
            current[key] = val
    new_cfg = AppConfig(**current)
    # Update singleton
    app_config.tools = new_cfg.tools
    app_config.stubs = new_cfg.stubs
    save_config(new_cfg)
    return {"status": "ok", "config": new_cfg.model_dump()}
```

## 9 — WEBSOCKET BROADCAST FUNCTION SIGNATURE

The `broadcast_fn` used throughout the stubs must match this signature
(consistent with the main prompt WebSocket protocol):

```python
# In backend/api/websocket.py

from typing import Optional

async def broadcast(
    step: str,
    level: str,
    message: str,
    progress: Optional[float] = None,
    data: Optional[dict] = None
):
    """
    Sends a WsMessage to all connected frontend WebSocket clients.
    step: "extract" | "rc" | "lfs" | "export" | "blender"
    level: "INFO" | "WARNING" | "ERROR" | "SUCCESS"
    """
    import json
    from datetime import datetime
    payload = {
        "type": "log" if data is None else "metric",
        "step": step,
        "timestamp": datetime.utcnow().isoformat(),
        "level": level,
        "message": message,
    }
    if progress is not None:
        payload["type"] = "progress"
        payload["progress"] = round(progress, 4)
    if data is not None:
        payload["type"] = "metric"
        payload["data"] = data
    # Send to all active WS connections
    for ws in active_connections:
        try:
            await ws.send_text(json.dumps(payload))
        except Exception:
            pass
```

## 10 — INTEGRATION IN `pipeline_runner.py`

```python
# backend/core/pipeline_runner.py  (excerpt)

from backend.core.steps.step_extract import run_extract, run_extract_stub
from backend.core.steps.step_rc import run_rc          # dispatcher included
from backend.core.steps.step_lfs import run_lfs        # dispatcher included
from backend.core.config import app_config
from backend.api.websocket import broadcast

async def run_pipeline(project_id: str, start_from_step: int = 1, settings: dict = {}):
    project_path = get_project_path(project_id)

    async def bcast(step, level, msg, progress=None, data=None):
        await broadcast(step, level, msg, progress, data)

    try:
        if start_from_step <= 2:
            if app_config.stubs.ffmpeg_stub:
                await run_extract_stub(project_path, bcast, settings)
            else:
                await run_extract(project_path, bcast, settings)

        if start_from_step <= 3:
            await run_rc(project_path, bcast, settings)  # dispatcher handles stub flag

        if start_from_step <= 4:
            await run_lfs(project_path, bcast, settings)  # dispatcher handles stub flag

        # ... step 5 export, step 6 blender (same pattern)

    except FileNotFoundError as e:
        await bcast("pipeline", "ERROR",
            f"Tool not found: {e}\n→ Enable stub mode in Settings to continue without this tool.")
    except RuntimeError as e:
        await bcast("pipeline", "ERROR", f"Pipeline error: {e}")
```

```markdown
## Development Without Hardware (Stub Mode)

By default, RealityCapture and LichtFeld Studio run in **stub/simulation mode**.
This means the full wizard pipeline can be tested without:
- An NVIDIA GPU
- RealityCapture installed
- LichtFeld Studio compiled from source

### What stubs do
- Stream realistic progressive log messages (same format as real tools)
- Generate valid output files (registration CSV, sparse PLY, trained PLY)
- Simulate realistic timing (8s for RC alignment, 15s for LFS training by default)
- Feed live metrics to the frontend charts (loss, PSNR, Gaussian count)

### How to disable stubs (production mode)
1. Build LichtFeld Studio: https://github.com/MrNeRF/LichtFeld-Studio/wiki
2. Install RealityCapture via Epic Games Launcher
3. Open the app → Settings → set exe paths
4. Toggle OFF "Stub Mode" in the RC and LFS advanced settings
5. Re-run the pipeline — real tools will be called

### Adjusting stub duration
In Settings > Advanced > DEV section:
- RC simulation: 3–60 seconds (default: 8s)
- LFS simulation: 3–60 seconds (default: 15s)
```
